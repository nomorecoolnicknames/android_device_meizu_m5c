# Proprietary blob manifest — Meizu M5c (m5c, M710H), MT6737M.

# Generated 2026-09-16 from vendor/meizu/m5c/m5c-vendor-blobs.mk, which is the
# FULL vendor-migration list produced by the LOS 16 lane on 2026-08-29.  That
# split is FACT-driven (a strings scan of every stock blob for hardcoded
# absolute paths), not stylistic:
#
#   -> vendor/bin:          all stock daemons and tools (including the former
#                           xbin/mnld); launch paths are owned by our rc files.
#   -> vendor/lib{,64}:     the MTK library mass, including egl/ (the EGL
#                           loader probes /vendor/lib*/egl first) and res/.
#   -> vendor/etc/firmware: modem images only (modem_*, dsp_*,
#                           catcher_filter_*, em_filter_*).  ccci_mdinit probes
#                           /custom/etc/firmware/ FIRST — the M-era daemon
#                           knows nothing of /vendor/firmware — and
#                           init.mt6735.rc makes /custom -> /vendor, so the
#                           stock probe order finds them.
#
#   STAYS IN /system, each for a hard reason:
#     - the RIL/MAL chain (mtk-ril*, librilmtk*, libmal*, volte_imsm,
#       libccci_util): mtkrild hardcodes /system/lib/mtk-ril.so and
#       libmal_imsmngr hardcodes /system/lib/volte_imsm.so (both
#       strings-verified).
#
# DELIBERATELY ABSENT (do NOT re-add when regenerating):
#   - hwcomposer.mt6737m.so — the device tree builds forge_hwc under the same
#     module name and PRODUCT_COPY_FILES silently beats a built module.
#     Shipping the blob replaced our composer, it died in
#     GameDetector::LayersStrategy, took SurfaceFlinger with it, and
#     system_server hung at "Starting phase 100" — black screen on a flashed
#     ROM.
#   - bin/md_ctrl — built against an older android_fork_execvp_ext signature;
#     on N+ it walks into __write_chk with a garbage count and dies on FORTIFY
#     ("write: count > SSIZE_MAX"), init respawns it, and the crash loop ends
#     in a watchdog reboot (WDT_STATUS 0x40000000 in pstore).  The device tree
#     carries md_ctrl sources instead.
#
#   - lib{,64}/hw/{audio_policy.stub,local_time.default,vibrator.default}.so
#     and lib{,64}/hw/keystore.default.so (removed 2026-09-16) — these are
#     MediaTek's 2014-toolchain build of AOSP sources, not MediaTek code: the
#     first three carry the verbatim hw_module_t .name of
#     hardware/libhardware/modules/{audio,local_time,vibrator} and Android 13
#     builds and installs its own current copies to the very same vendor paths,
#     where PRODUCT_COPY_FILES would silently overwrite them.  keystore.default
#     is AOSP softkeymaster and its DT_NEEDED libs no longer exist in A13, so it
#     cannot load at all.  power.default.so is NOT in this group and stays: its
#     .name is "Legacy PowerHAL" and it drives /proc/hps/power_mode — real MTK.
#
